Ingress for our code 

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: springboot-multi-ingress
  namespace: default
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /$2
spec:
  ingressClassName: nginx
  rules:
  - host: "springboot.local"
    http:
      paths:
      - path: /api1(/|$)(.*)
        pathType: ImplementationSpecific  # Changed from Prefix
        backend:
          service:
            name: api1-service
            port:
              number: 80
      - path: /api2(/|$)(.*)
        pathType: ImplementationSpecific  # Changed from Prefix
        backend:
          service:
            name: api2-service
            port:
              number: 80
